async function analizarSentimientoIA(texto) {
  if (!hf) {
    console.warn('⚠️ No hay token HF, usando fallback para sentimiento');
    return analizarSentimientoFallback(texto);
  }
  try {
    const resultado = await hf.textClassification({
      model: 'finiteautomata/beto-sentiment-analysis',
      inputs: texto.substring(0, 500),
    });
    const label = resultado[0].label;
    const score = resultado[0].score;
    if (label === 'POS') return score;
    if (label === 'NEG') return -score;
    return 0;
  } catch (error) {
    console.warn('⚠️ Error en IA de sentimiento, usando fallback:', error.message);
    return analizarSentimientoFallback(texto);
  }
}

function analizarSentimientoFallback(texto) {
  const analyzer = new natural.SentimentAnalyzer('Spanish', natural.PorterStemmerEs, 'afinn');
  const tokenizer = new natural.WordTokenizer();
  const tokens = tokenizer.tokenize(texto);
  const score = analyzer.getSentiment(tokens);
  return Math.max(-1, Math.min(1, score / 5));
}

function detectarPostura(texto, tema) {
  const config = POSTURAS_POR_TEMA[tema];
  if (!config) return { postura: 'neutral', score: 0 };
  const textoLower = texto.toLowerCase();
  let scoreA = 0, scoreB = 0;
  config.postura_a.forEach(p => { if (textoLower.includes(p)) scoreA++; });
  config.postura_b.forEach(p => { if (textoLower.includes(p)) scoreB++; });
  if (scoreA > scoreB) return { postura: config.nombre_a, score: scoreA / (scoreA + scoreB + 1) };
  if (scoreB > scoreA) return { postura: config.nombre_b, score: scoreB / (scoreA + scoreB + 1) };
  return { postura: 'neutral', score: 0 };
}

async function detectarContradiccionesIA(noticia, todasLasNoticias) {
  let temas = [];
  try { temas = JSON.parse(noticia.temas_detectados || '[]'); } catch { temas = await detectarTemasIA(noticia.titular, noticia.descripcion); }
  if (temas.length === 0) return [];
  
  const contradicciones = [];
  const fechaNoticia = new Date(noticia.fecha);
  const noticiasMismoDia = todasLasNoticias.filter(n => {
    if (n.id === noticia.id) return false;
    return Math.abs(new Date(n.fecha) - fechaNoticia) < 24 * 60 * 60 * 1000;
  });
  
  for (const { tema } of temas) {
    const posturaNoticia = detectarPostura(noticia.titular + ' ' + noticia.descripcion, tema);
    if (posturaNoticia.postura === 'neutral') continue;
    for (const otraNoticia of noticiasMismoDia) {
      const posturaOtra = detectarPostura(otraNoticia.titular + ' ' + otraNoticia.descripcion, tema);
      if (posturaOtra.postura === 'neutral') continue;
      if (posturaOtra.postura === posturaNoticia.postura) continue;
      const intensidad = (posturaNoticia.score + posturaOtra.score) / 2;
      contradicciones.push({
        noticia_contraria_id: otraNoticia.id,
        noticia_contraria_titular: otraNoticia.titular,
        noticia_contraria_fuente: otraNoticia.fuente,
        tema, postura_a: posturaNoticia.postura, postura_b: posturaOtra.postura, intensidad
      });
    }
  }
  return contradicciones.sort((a, b) => b.intensidad - a.intensidad).slice(0, 5);
}

async function recolectarDeRSS(fuente) {
  try {
    console.log(`  📡 Recolectando de ${fuente.nombre}...`);
    const feed = await rssParser.parseURL(fuente.url);
    return feed.items.map(item => ({
      id: `${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
      titular: item.title?.trim() || 'Sin título',
      descripcion: item.contentSnippet?.trim() || item.content?.replace(/<[^>]*>/g, '').trim().substring(0, 300) || '',
      enlace: item.link || '',
      fecha: item.pubDate ? new Date(item.pubDate).toISOString() : new Date().toISOString(),
      fuente: fuente.nombre,
      color_fuente: fuente.color,
      imagen: item.enclosure?.url || null,
    }));
  } catch (error) {
    console.error(`  ❌ Error con ${fuente.nombre}:`, error.message);
    return [];
  }
}

async function procesarNoticias(noticiasCrudas) {
  console.log('\n🔍 ANALIZANDO NOTICIAS...');
  console.log('═'.repeat(60));
  
  const noticiasProcesadas = [];
  
  for (let i = 0; i < noticiasCrudas.length; i++) {
    const noticia = noticiasCrudas[i];
    if (i % 10 === 0) console.log(`  Procesando... ${i}/${noticiasCrudas.length}`);
    
    noticia.sentimiento = await analizarSentimientoIA(noticia.titular + ' ' + noticia.descripcion);
    const temas = await detectarTemasIA(noticia.titular, noticia.descripcion);
    noticia.temas_detectados = JSON.stringify(temas);
    noticia.intensidad_contradiccion = 0;
    noticia.eje_z = 0;
    noticia.contradicciones_ids = '[]';
    
    insertNoticia.run(noticia.id, noticia.titular, noticia.descripcion, noticia.enlace,
      noticia.fecha, noticia.fuente, noticia.color_fuente, noticia.imagen,
      noticia.sentimiento, 0, 0, noticia.temas_detectados, '[]');
    
    noticiasProcesadas.push(noticia);
    if (i % 5 === 0) await new Promise(resolve => setTimeout(resolve, 500));
  }
  
  console.log('\n🔍 DETECTANDO CONTRADICCIONES...');
  let totalContradicciones = 0;
  
  for (let i = 0; i < noticiasProcesadas.length; i++) {
    const noticia = noticiasProcesadas[i];
    const contradicciones = await detectarContradiccionesIA(noticia, noticiasProcesadas);
    
    if (contradicciones.length > 0) {
      const intensidadPromedio = contradicciones.reduce((sum, c) => sum + c.intensidad, 0) / contradicciones.length;
      
      db.prepare(`UPDATE noticias SET intensidad_contradiccion = ?, eje_z = ?, contradicciones_ids = ?, actualizado_en = datetime('now') WHERE id = ?`)
        .run(intensidadPromedio, intensidadPromedio * 3, JSON.stringify(contradicciones.map(c => c.noticia_contraria_id)), noticia.id);
      
      const insertContrad = db.prepare('INSERT INTO contradicciones (noticia_id, noticia_contraria_id, tema, intensidad) VALUES (?, ?, ?, ?)');
      contradicciones.forEach(c => insertContrad.run(noticia.id, c.noticia_contraria_id, c.tema, c.intensidad));
      totalContradicciones += contradicciones.length;
    }
    if (i % 20 === 0) console.log(`  Detectando... ${i}/${noticiasProcesadas.length}`);
  }
  
  console.log('═'.repeat(60));
  console.log(`✅ ${totalContradicciones} contradicciones detectadas`);
  return noticiasProcesadas;
}

  const material = new THREE.MeshStandardMaterial({ map: textura, roughness: 0.3, metalness: 0.2, side: THREE.DoubleSide });
  const carta = new THREE.Mesh(geometria, material);
  
  const x = indice * CONFIG.espacioX;
  const y = Math.sin(indice * 0.5) * 0.3;
  const z = noticia.eje_z || 0;
  const angulo = x / CONFIG.curvatura;
  const xCurvado = CONFIG.curvatura * Math.sin(angulo);
  const zCurvado = CONFIG.curvatura * (1 - Math.cos(angulo)) + z;
  
  carta.position.set(xCurvado, y, zCurvado);
  carta.rotation.y = -angulo;
  carta.userData = { noticia, indice, posicionOriginal: { x: xCurvado, y, z: zCurvado } };
  
  const bordeGeo = new THREE.EdgesGeometry(geometria);
  const colorBorde = noticia.intensidad_contradiccion > 0 ? 0xff4444 : 0x334466;
  const borde = new THREE.LineSegments(bordeGeo, new THREE.LineBasicMaterial({ color: colorBorde }));
  carta.add(borde);
  
  return carta;
}

function detectarTemaYPostura(titular, descripcion = '') {
  const texto = (titular + ' ' + descripcion).toLowerCase();
  const resultados = [];
  
  for (const [tema, posturas] of Object.entries(TEMAS_CONTRADICTORIOS)) {
    let scorePro = 0;
    let scoreContra = 0;
    
    posturas.pro.forEach(palabra => {
      if (texto.includes(palabra)) scorePro++;
    });
    
    posturas.contra.forEach(palabra => {
      if (texto.includes(palabra)) scoreContra++;
    });
    
    if (scorePro > 0 || scoreContra > 0) {
      resultados.push({
        tema,
        scorePro,
        scoreContra,
        polarizacion: Math.abs(scorePro - scoreContra) / (scorePro + scoreContra + 1)
      });
    }
  }
  
  return resultados;
}

function detectarContradicciones(noticia, todasLasNoticias) {
  const analisisNoticia = detectarTemaYPostura(noticia.titular, noticia.descripcion);
  
  if (analisisNoticia.length === 0) return [];
  
  const contradicciones = [];
  
  // Buscar noticias del mismo día que traten temas similares con postura opuesta
  const noticiasMismoDia = todasLasNoticias.filter(n => {
    if (n._id === noticia._id || n._id?.toString() === noticia._id?.toString()) return false;
    const fechaN = new Date(n.fecha);
    const fechaNoticia = new Date(noticia.fecha);
    return Math.abs(fechaN - fechaNoticia) < 24 * 60 * 60 * 1000; // 24 horas
  });
  
  analisisNoticia.forEach(analisis => {
    noticiasMismoDia.forEach(otraNoticia => {
      const analisisOtra = detectarTemaYPostura(otraNoticia.titular, otraNoticia.descripcion);
      
      analisisOtra.forEach(ao => {
        if (ao.tema === analisis.tema) {
          // Una es pro y otra contra
          const esContradiccion = 
            (analisis.scorePro > 0 && ao.scoreContra > 0) ||
            (analisis.scoreContra > 0 && ao.scorePro > 0);
          
          if (esContradiccion) {
            contradicciones.push({
              noticiaContraria: otraNoticia,
              tema: analisis.tema,
              intensidad: Math.max(analisis.polarizacion, ao.polarizacion)
            });
          }
        }
      });
    });
  });
  
  return contradicciones;
}

    size: 0.05,

    transparent: true,
    blending: THREE.AdditiveBlending,
    depthWrite: false,

// ==================== CREAR TEXTURA DE NOTICIA ====================
function crearTexturaNoticia(noticia) {
  const canvas = document.createElement('canvas');
  canvas.width = 512;
  canvas.height = 358;
  const ctx = canvas.getContext('2d');
  
  // Fondo
  ctx.fillStyle = '#0f0f2e';
  ctx.fillRect(0, 0, 512, 358);
  
  // Borde con color de la fuente
  ctx.strokeStyle = noticia.colorFuente || '#4466aa';
  ctx.lineWidth = 4;
  ctx.strokeRect(10, 10, 492, 338);
  
  // Indicador de contradicción
  if (noticia.intensidadContradiccion > 0) {
    ctx.fillStyle = `rgba(255, 80, 80, ${noticia.intensidadContradiccion * 0.3})`;
    ctx.fillRect(10, 10, 492, 338);
    
    // Borde pulsante
    ctx.strokeStyle = '#ff4444';
    ctx.lineWidth = 6;
    ctx.strokeRect(10, 10, 492, 338);
  }
  
  // Título (máximo 3 líneas)
  ctx.fillStyle = '#ffffff';
  ctx.font = 'bold 26px "Segoe UI", Arial';
  ctx.textAlign = 'left';
  
  const palabras = noticia.titular.split(' ');
  let lineas = [];
  let lineaActual = '';
  
  palabras.forEach(palabra => {
    const prueba = lineaActual + palabra + ' ';
    if (ctx.measureText(prueba).width > 460) {
      lineas.push(lineaActual.trim());
      lineaActual = palabra + ' ';
    } else {
      lineaActual = prueba;
    }
  });
  lineas.push(lineaActual.trim());
  
  lineas = lineas.slice(0, 3);
  
  lineas.forEach((linea, i) => {
    ctx.fillText(linea, 30, 80 + i * 42);
  });
  
  // Fuente
  ctx.fillStyle = noticia.colorFuente || '#aaa';
  ctx.font = 'bold 20px "Segoe UI", Arial';
  ctx.fillText(noticia.fuente, 30, 240);
  
  // Fecha
  ctx.fillStyle = '#888';
  ctx.font = '16px "Segoe UI", Arial';
  const fecha = new Date(noticia.fecha);
  ctx.fillText(fecha.toLocaleDateString('es-ES', { 
    day: 'numeric', 
    month: 'short', 
    hour: '2-digit', 
    minute: '2-digit' 
  }), 30, 275);
  
  // Badge de contradicción
  if (noticia.intensidadContradiccion > 0) {
    ctx.fillStyle = '#ff4444';
    ctx.font = 'bold 18px "Segoe UI", Arial';

  }

}

// ==================== CREAR CARTA 3D ====================
function crearCarta(noticia, indice) {
  const geometria = new THREE.PlaneGeometry(CONFIG.anchoCarta, CONFIG.altoCarta);
  
  const textura = crearTexturaNoticia(noticia);
  textura.colorSpace = THREE.SRGBColorSpace;
  
  const material = new THREE.MeshStandardMaterial({
    map: textura,
    roughness: 0.3,
    metalness: 0.2,
    side: THREE.DoubleSide,
  });
  
  const carta = new THREE.Mesh(geometria, material);
  
  // Posición en la cronología horizontal
  const x = indice * CONFIG.espacioX;
  const y = (Math.sin(indice * 0.5) * 0.3); // Ligera ondulación vertical
  const z = noticia.ejeZ || 0; // ¡Aquí sobresalen las contradictorias!
  
  // Aplicar curvatura suave
  const angulo = x / CONFIG.curvatura;
  const xCurvado = CONFIG.curvatura * Math.sin(angulo);
  const zCurvado = CONFIG.curvatura * (1 - Math.cos(angulo)) + z;
  
  carta.position.set(xCurvado, y, zCurvado);
  carta.rotation.y = -angulo;
  
  // Guardar datos
  carta.userData = {
    noticia,
    indice,
    posicionOriginal: { x: xCurvado, y, z: zCurvado }
  };
  
  // Borde
  const bordeGeo = new THREE.EdgesGeometry(geometria);
  const colorBorde = noticia.intensidadContradiccion > 0 ? 0xff4444 : 0x334466;
  const bordeMat = new THREE.LineBasicMaterial({ 
    color: colorBorde,
    linewidth: 1,
  });
  const borde = new THREE.LineSegments(bordeGeo, bordeMat);
  carta.add(borde);
  
  return carta;
}

// ==================== ACTUALIZAR MURO ====================
function actualizarMuro() {
  // Limpiar cartas existentes
  while (grupoMuro.children.length > 0) {
    const carta = grupoMuro.children[0];
    carta.material.map?.dispose();
    carta.material.dispose();
    carta.geometry.dispose();
    grupoMuro.remove(carta);
  }
  
  cartasEnEscena = [];
  
  // Crear cartas para las noticias visibles
  noticiasVisibles.forEach((noticia, i) => {
    const carta = crearCarta(noticia, i);
    grupoMuro.add(carta);
    cartasEnEscena.push(carta);
  });
  
  // Actualizar estadísticas
  document.getElementById('total-noticias').textContent = todasLasNoticias.length;
  document.getElementById('total-contradictorias').textContent = 
    todasLasNoticias.filter(n => n.intensidadContradiccion > 0).length;
  
  // Ocultar barra de carga
  document.getElementById('progreso-carga').style.width = '100%';
  setTimeout(() => {
    document.getElementById('barra-carga').style.opacity = '0';
  }, 500);
}

// ==================== CARGAR NOTICIAS DEL BACKEND ====================
async function cargarNoticias(pagina = 1) {
  if (cargando) return;
  cargando = true;
  
  document.getElementById('barra-carga').style.opacity = '1';
  document.getElementById('progreso-carga').style.width = '0%';
  
  try {
    const respuesta = await fetch(`/api/noticias?pagina=${pagina}&porPagina=${CONFIG.cargaPorLote}`);
    const datos = await respuesta.json();
    
    document.getElementById('progreso-carga').style.width = '50%';
    
    if (datos.noticias.length === 0) {
      hayMasNoticias = false;
      console.log('🏁 No hay más noticias disponibles');
    } else {
      todasLasNoticias = [...todasLasNoticias, ...datos.noticias];
      
      // Eliminar duplicados por _id
      const ids = new Set();
      todasLasNoticias = todasLasNoticias.filter(n => {
        if (ids.has(n._id)) return false;
        ids.add(n._id);
        return true;
      });
      
      // Ordenar por fecha (más reciente = índice bajo = más a la derecha）
      todasLasNoticias.sort((a, b) => new Date(b.fecha) - new Date(a.fecha));
      
      paginaActual = pagina;
      hayMasNoticias = datos.noticias.length === CONFIG.cargaPorLote;
      
      document.getElementById('progreso-carga').style.width = '75%';
    }
    
    // Actualizar noticias visibles (últimas N)
    noticiasVisibles = todasLasNoticias.slice(0, CONFIG.maxNoticiasVisibles);
    
    actualizarMuro();
    
    console.log(`📰 ${todasLasNoticias.length} noticias en total, ${noticiasVisibles.length} visibles`);
  } catch (error) {
    console.error('❌ Error cargando noticias:', error);
    document.getElementById('barra-carga').style.opacity = '0';

// Modelo de análisis de sentimiento en español
async function analizarSentimientoIA(texto) {
  try {
    const resultado = await hf.textClassification({
      model: 'finiteautomata/beto-sentiment-analysis',
      inputs: texto.substring(0, 500), // Máximo 500 caracteres
    });
    
    // Convertir a score -1 a 1
    const label = resultado[0].label;
    const score = resultado[0].score;
    
    if (label === 'POS') return score;
    if (label === 'NEG') return -score;
    return 0;
  } catch (error) {
    console.warn('⚠️ Error en IA de sentimiento, usando fallback:', error.message);
    return analizarSentimientoFallback(texto);
  }
}

// Fallback: análisis basado en reglas
function analizarSentimientoFallback(texto) {
  const analyzer = new natural.SentimentAnalyzer('Spanish', natural.PorterStemmerEs, 'afinn');
  const tokenizer = new natural.WordTokenizer();
  const tokens = tokenizer.tokenize(texto);
  const score = analyzer.getSentiment(tokens);
  return Math.max(-1, Math.min(1, score / 5));
}

// Modelo de detección de temas con zero-shot classification
async function detectarTemasIA(titular, descripcion) {
  const texto = (titular + ' ' + descripcion).substring(0, 500);
  
  const temasCandidatos = [
    'política nacional',
    'economía y finanzas',
    'inmigración y fronteras',
    'vivienda y urbanismo',
    'cataluña y independentismo',
    'sanidad y salud pública',
    'educación',
    'medio ambiente y cambio climático',
    'tecnología e innovación',
    'deportes',
    'conflicto internacional',
    'derechos sociales',
    'corrupción',
    'seguridad ciudadana',
    'Unión Europea'
  ];
  
  try {
    const resultado = await hf.zeroShotClassification({
      model: 'MoritzLaurer/mDeBERTa-v3-base-xnli-multilingual-nli-2mil7',
      inputs: texto,
      parameters: {
        candidate_labels: temasCandidatos,
        multi_label: true,
      }
    });
    
    // Devolver temas con score > 0.3
    return resultado.labels
      .filter((_, i) => resultado.scores[i] > 0.3)
      .map((label, i) => ({
        tema: label,
        score: resultado.scores[i]
      }));
  } catch (error) {
    console.warn('⚠️ Error en IA de temas, usando fallback:', error.message);
    return detectarTemasFallback(titular, descripcion);
  }
}

function detectarPostura(texto, tema) {
  const config = POSTURAS_POR_TEMA[tema];
  if (!config) return { postura: 'neutral', score: 0 };
  
  const textoLower = texto.toLowerCase();
  let scoreA = 0;
  let scoreB = 0;
  
  config.postura_a.forEach(p => {
    if (textoLower.includes(p)) scoreA++;
  });
  
  config.postura_b.forEach(p => {
    if (textoLower.includes(p)) scoreB++;
  });
  
  if (scoreA > scoreB) return { postura: config.nombre_a, score: scoreA / (scoreA + scoreB + 1) };
  if (scoreB > scoreA) return { postura: config.nombre_b, score: scoreB / (scoreA + scoreB + 1) };
  return { postura: 'neutral', score: 0 };
}

async function detectarContradiccionesIA(noticia, todasLasNoticias) {
  // Obtener temas de la noticia
  let temas = [];
  try {
    const temasGuardados = JSON.parse(noticia.temas_detectados || '[]');
    temas = temasGuardados;
  } catch {
    temas = await detectarTemasIA(noticia.titular, noticia.descripcion);
  }
  
  if (temas.length === 0) return [];
  
  const contradicciones = [];
  
  // Buscar noticias del mismo día
  const fechaNoticia = new Date(noticia.fecha);
  const noticiasMismoDia = todasLasNoticias.filter(n => {
    if (n.id === noticia.id) return false;
    const fechaN = new Date(n.fecha);
    return Math.abs(fechaN - fechaNoticia) < 24 * 60 * 60 * 1000;
  });
  
  // Para cada tema detectado, buscar posturas opuestas
  for (const { tema, score: scoreTema } of temas) {
    const posturaNoticia = detectarPostura(noticia.titular + ' ' + noticia.descripcion, tema);
    
    if (posturaNoticia.postura === 'neutral') continue;
    
    for (const otraNoticia of noticiasMismoDia) {
      const posturaOtra = detectarPostura(otraNoticia.titular + ' ' + otraNoticia.descripcion, tema);
      
      if (posturaOtra.postura === 'neutral') continue;
      if (posturaOtra.postura === posturaNoticia.postura) continue;
      
      // ¡Contradicción encontrada!
      const intensidad = (posturaNoticia.score + posturaOtra.score + scoreTema) / 3;
      
      contradicciones.push({
        noticia_contraria_id: otraNoticia.id,
        noticia_contraria_titular: otraNoticia.titular,
        noticia_contraria_fuente: otraNoticia.fuente,
        tema,
        postura_a: posturaNoticia.postura,
        postura_b: posturaOtra.postura,
        intensidad
      });
    }
  }
  
  // Ordenar por intensidad y limitar
  return contradicciones
    .sort((a, b) => b.intensidad - a.intensidad)
    .slice(0, 5);
}

async function procesarNoticias(noticiasCrudas) {
  console.log('\n🔍 ANALIZANDO NOTICIAS...');
  console.log('═'.repeat(60));
  
  const noticiasProcesadas = [];
  const todasLasIds = noticiasCrudas.map(n => n.id);
  
  for (let i = 0; i < noticiasCrudas.length; i++) {
    const noticia = noticiasCrudas[i];
    
    // Progreso
    if (i % 10 === 0) {
      console.log(`  Procesando... ${i}/${noticiasCrudas.length}`);
    }
    
    // 1. Análisis de sentimiento con IA
    noticia.sentimiento = await analizarSentimientoIA(
      noticia.titular + ' ' + noticia.descripcion
    );
    
    // 2. Detección de temas con IA
    const temas = await detectarTemasIA(noticia.titular, noticia.descripcion);
    noticia.temas_detectados = JSON.stringify(temas);
    
    // 3. Guardar en base de datos (sin contradicciones aún)
    noticia.intensidad_contradiccion = 0;
    noticia.eje_z = 0;
    noticia.contradicciones_ids = '[]';
    
    insertNoticia.run(
      noticia.id, noticia.titular, noticia.descripcion, noticia.enlace,
      noticia.fecha, noticia.fuente, noticia.color_fuente, noticia.imagen,
      noticia.sentimiento, noticia.intensidad_contradiccion, noticia.eje_z,
      noticia.temas_detectados, noticia.contradicciones_ids
    );
    
    noticiasProcesadas.push(noticia);
    
    // Pequeña pausa para no saturar la API de Hugging Face
    if (i % 5 === 0) {
      await new Promise(resolve => setTimeout(resolve, 500));
    }
  }
  
  // 4. Detección de contradicciones (segunda pasada)
  console.log('\n🔍 DETECTANDO CONTRADICCIONES...');
  let totalContradicciones = 0;
  
  for (let i = 0; i < noticiasProcesadas.length; i++) {
    const noticia = noticiasProcesadas[i];
    
    const contradicciones = await detectarContradiccionesIA(noticia, noticiasProcesadas);
    
    if (contradicciones.length > 0) {
      const intensidadPromedio = contradicciones.reduce((sum, c) => sum + c.intensidad, 0) / contradicciones.length;
      
      // Actualizar en BD
      db.prepare(`
        UPDATE noticias 
        SET intensidad_contradiccion = ?, 
            eje_z = ?, 
            contradicciones_ids = ?,
            actualizado_en = datetime('now')
        WHERE id = ?
      `).run(
        intensidadPromedio,
        intensidadPromedio * 3,
        JSON.stringify(contradicciones.map(c => c.noticia_contraria_id)),
        noticia.id
      );
      
      // Guardar contradicciones en tabla
      const insertContrad = db.prepare('INSERT INTO contradicciones (noticia_id, noticia_contraria_id, tema, intensidad) VALUES (?, ?, ?, ?)');
      contradicciones.forEach(c => {
        insertContrad.run(noticia.id, c.noticia_contraria_id, c.tema, c.intensidad);
      });
      
      totalContradicciones += contradicciones.length;
    }
    
    if (i % 20 === 0) {
      console.log(`  Detectando contradicciones... ${i}/${noticiasProcesadas.length}`);
    }
  }
  
  console.log('═'.repeat(60));
  console.log(`✅ ${totalContradicciones} contradicciones detectadas en total`);
  
  return noticiasProcesadas;
}

async function analizarSentimientoIA(texto) {
  if (!hf) {
    console.warn('⚠️ No hay token HF, usando fallback para sentimiento');
    return analizarSentimientoFallback(texto);
  }
  try {
    const resultado = await hf.textClassification({
      model: 'finiteautomata/beto-sentiment-analysis',
      inputs: texto.substring(0, 500),
    });
    const label = resultado[0].label;
    const score = resultado[0].score;
    if (label === 'POS') return score;
    if (label === 'NEG') return -score;
    return 0;
  } catch (error) {
    console.warn('⚠️ Error en IA de sentimiento, usando fallback:', error.message);
    return analizarSentimientoFallback(texto);
  }
}

function analizarSentimientoFallback(texto) {
  const analyzer = new natural.SentimentAnalyzer('Spanish', natural.PorterStemmerEs, 'afinn');
  const tokenizer = new natural.WordTokenizer();
  const tokens = tokenizer.tokenize(texto);
  const score = analyzer.getSentiment(tokens);
  return Math.max(-1, Math.min(1, score / 5));
}

function detectarPostura(texto, tema) {
  const config = POSTURAS_POR_TEMA[tema];
  if (!config) return { postura: 'neutral', score: 0 };
  const textoLower = texto.toLowerCase();
  let scoreA = 0, scoreB = 0;
  config.postura_a.forEach(p => { if (textoLower.includes(p)) scoreA++; });
  config.postura_b.forEach(p => { if (textoLower.includes(p)) scoreB++; });
  if (scoreA > scoreB) return { postura: config.nombre_a, score: scoreA / (scoreA + scoreB + 1) };
  if (scoreB > scoreA) return { postura: config.nombre_b, score: scoreB / (scoreA + scoreB + 1) };
  return { postura: 'neutral', score: 0 };
}

async function detectarContradiccionesIA(noticia, todasLasNoticias) {
  let temas = [];
  try { temas = JSON.parse(noticia.temas_detectados || '[]'); } catch { temas = await detectarTemasIA(noticia.titular, noticia.descripcion); }
  if (temas.length === 0) return [];
  
  const contradicciones = [];
  const fechaNoticia = new Date(noticia.fecha);
  const noticiasMismoDia = todasLasNoticias.filter(n => {
    if (n.id === noticia.id) return false;
    return Math.abs(new Date(n.fecha) - fechaNoticia) < 24 * 60 * 60 * 1000;
  });
  
  for (const { tema } of temas) {
    const posturaNoticia = detectarPostura(noticia.titular + ' ' + noticia.descripcion, tema);
    if (posturaNoticia.postura === 'neutral') continue;
    for (const otraNoticia of noticiasMismoDia) {
      const posturaOtra = detectarPostura(otraNoticia.titular + ' ' + otraNoticia.descripcion, tema);
      if (posturaOtra.postura === 'neutral') continue;
      if (posturaOtra.postura === posturaNoticia.postura) continue;
      const intensidad = (posturaNoticia.score + posturaOtra.score) / 2;
      contradicciones.push({
        noticia_contraria_id: otraNoticia.id,
        noticia_contraria_titular: otraNoticia.titular,
        noticia_contraria_fuente: otraNoticia.fuente,
        tema, postura_a: posturaNoticia.postura, postura_b: posturaOtra.postura, intensidad
      });
    }
  }
  return contradicciones.sort((a, b) => b.intensidad - a.intensidad).slice(0, 5);
}
