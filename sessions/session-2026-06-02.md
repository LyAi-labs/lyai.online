import { View, Text, TextInput, TouchableOpacity, FlatList, StyleSheet, ActivityIndicator } from 'react-native';

import { api } from '../utils/api';

const PDS_STATION_IDS = [
  'champery', 'les_crosets', 'champoussin', 'morgins', 'torgon',
  'avoriaz', 'morzine', 'les_gets', 'chatel', 'abondance', 'la_chapelle', 'st_jean_aulps',
];

export type EndpointKind = 'station' | 'piste' | 'lift';

export interface RouteEndpoint {

  label: string;
  stationId?: string;
  lat: number;
  lng: number;
  entityId?: string;
}

interface PisteResult { piste_id: string; name: string; difficulty?: string; station_id?: string; lat?: number; lng?: number; }
interface LiftResult { slug: string; name?: string; subtipo?: string; station_id?: string; station_name?: string; lat?: number; lng?: number; }

interface Props {
  label: string;
  value: RouteEndpoint | null;
  onChange: (v: RouteEndpoint | null) => void;
  stationHint?: string;
  optional?: boolean;
}

const KINDS: { id: EndpointKind; label: string; icon: any }[] = [
  { id: 'station', label: 'Estación', icon: 'business' },
  { id: 'piste',   label: 'Pista',    icon: 'trail-sign' },
  { id: 'lift',    label: 'Remonte',  icon: 'arrow-up-circle' },
];

export default function RouteEndpointPicker({ label, value, onChange, stationHint, optional }: Props) {
  const { stations } = useApp();
  const [activeKind, setActiveKind] = useState<EndpointKind>(value?.kind || 'station');
  const [query, setQuery] = useState('');
  const [results, setResults] = useState<(PisteResult | LiftResult)[]>([]);
  const [searching, setSearching] = useState(false);
  const [showDropdown, setShowDropdown] = useState(false);
  const debounceTimer = useRef<any>(null);

  const pdsStations = useMemo(() => {
    const all = (stations || []).filter((s: any) => PDS_STATION_IDS.includes(s.station_id));
    return {
      france: all.filter((s: any) => s.country === 'France'),
      swiss:  all.filter((s: any) => s.country !== 'France'),
    };
  }, [stations]);

  useEffect(() => {
    if (debounceTimer.current) clearTimeout(debounceTimer.current);
    if (activeKind !== 'piste' && activeKind !== 'lift') { setResults([]); setShowDropdown(false); return; }
    debounceTimer.current = setTimeout(async () => {
      setSearching(true);
      try {
        const hint = (value?.kind === 'station' ? value.stationId : undefined) || stationHint;
        const res = activeKind === 'piste'
          ? await api.searchPistes(query, hint)
          : await api.searchLifts(query, hint);
        setResults(Array.isArray(res) ? res : []);
        setShowDropdown(true);
      } catch { setResults([]); }
      finally { setSearching(false); }
    }, query ? 250 : 0);
    return () => debounceTimer.current && clearTimeout(debounceTimer.current);
  }, [query, activeKind, stationHint, value?.kind, value?.stationId]);

  const switchKind = (k: EndpointKind) => {
    if (k === activeKind) return;
    setActiveKind(k); setQuery(''); setResults([]); setShowDropdown(false);
  };

  const pickStation = (st: any) => {
    if (st.lat == null || st.lng == null) return;
    onChange({ kind: 'station', label: st.name, stationId: st.station_id, lat: st.lat, lng: st.lng });
    setShowDropdown(false);
  };
  const pickPiste = (p: PisteResult) => {
    if (p.lat == null || p.lng == null) return;
    onChange({ kind: 'piste', label: p.name, stationId: p.station_id, lat: p.lat, lng: p.lng, entityId: p.piste_id });
    setQuery(p.name); setShowDropdown(false);
  };
  const pickLift = (l: LiftResult) => {
    if (l.lat == null || l.lng == null) return;
    onChange({ kind: 'lift', label: l.name || l.slug, stationId: l.station_id, lat: l.lat, lng: l.lng, entityId: l.slug });
    setQuery(l.name || l.slug); setShowDropdown(false);
  };
  const clear = () => { onChange(null); setQuery(''); };

  const grouped = useMemo(() => {

    if (!hint || activeKind === 'station') return { priority: results, others: [] as any[] };

        <Text style={s.label}>{label}</Text>

        })}

                <Text style={s.stationAvatarText}>{(value.label || '?').charAt(0).toUpperCase()}</Text>
              </View>
              <Text style={s.stationName} numberOfLines={1}>{value.label}</Text>
              <Ionicons name="chevron-down" size={16} color="#94A3B8" />
            </View>
          ) : (
            <View style={s.stationCardEmpty}>
              <Ionicons name="add" size={16} color="#94A3B8" />
              <Text style={s.stationCardEmptyText}>
                {optional ? 'Añadir parada (opcional)' : '— Seleccionar estación —'}
              </Text>
            </View>
          )}
          <View style={s.stationGrid}>
            {[...pdsStations.france, ...pdsStations.swiss].map((st: any) => {
              const sel = value?.kind === 'station' && value.stationId === st.station_id;
              return (
                <TouchableOpacity key={st.station_id} onPress={() => pickStation(st)}
                  style={[s.stationPill, sel && s.stationPillActive]}>
                  <Text style={[s.stationPillText, sel && s.stationPillTextActive]}>
                    {st.country === 'France' ? '🇫🇷' : '🇨🇭'} {st.name}
                  </Text>
                </TouchableOpacity>
              );
            })}
          </View>
        </View>
      )}

      {(activeKind === 'piste' || activeKind === 'lift') && (

            <Ionicons name="search" size={16} color="#878787" />
            <TextInput style={s.searchInput}

              placeholderTextColor="#94A3B8"
              value={query} onChangeText={setQuery} onFocus={() => setShowDropdown(true)} />
            {searching && <ActivityIndicator size="small" color="#878787" />}

                <Ionicons name="close-circle" size={16} color="#878787" />
              </TouchableOpacity>
            )}
          </View>
          {value && (value.kind === activeKind) && (
            <View style={s.pickedRow}>
              <Ionicons name="checkmark-circle" size={12} color="#10B981" />
              <Text style={s.pickedText} numberOfLines={1}>{value.label}</Text>
            </View>
          )}
          {showDropdown && (results.length > 0 || searching) && (

              {grouped.priority.length > 0 && grouped.others.length > 0 && (

                </Text>

                </>
              )}
            </View>
          )}
        </View>
      )}
    </View>
  );

function stationAvatarColor(stationId?: string): string {
  const palette = ['#475569', '#1A253D', '#64748B', '#2563EB', '#0D9488', '#7C3AED', '#B91C1C', '#D97706', '#16A34A', '#1E40AF', '#0EA5E9', '#94A3B8'];
  if (!stationId) return palette[0];
  let h = 0; for (let i = 0; i < stationId.length; i++) h = (h * 31 + stationId.charCodeAt(i)) | 0;
  return palette[Math.abs(h) % palette.length];
}

function renderResult(item: any, kind: EndpointKind, pickPiste: (p: any) => void, pickLift: (l: any) => void) {
  if (kind === 'piste') {
    const difColor = item.difficulty === 'black' ? '#0A0A0A'
                   : item.difficulty === 'red' ? '#B91C1C'
                   : item.difficulty === 'blue' ? '#2563EB' : '#10B981';
    return (
      <TouchableOpacity onPress={() => pickPiste(item)} style={s.resultItem}>
        <View style={[s.difDot, { backgroundColor: difColor }]} />
        <Text style={s.resultName} numberOfLines={1}>{item.name}</Text>
        <Text style={s.resultStation}>{item.station_id}</Text>
      </TouchableOpacity>
    );
  }
  return (
    <TouchableOpacity onPress={() => pickLift(item)} style={s.resultItem}>
      <Ionicons name="arrow-up-circle" size={12} color="#3C6EA1" />
      <Text style={s.resultName} numberOfLines={1}>{item.name || item.slug}</Text>
      <Text style={s.resultStation}>{item.station_name || item.station_id}</Text>
    </TouchableOpacity>
  );
}
