### 1. `~/.zshrc`

```bash
# ============================================================
# GLADO // neural shell
# ============================================================

# ------------------------------------------------------------
# Starship
# ------------------------------------------------------------

eval "$(starship init zsh)"

# ------------------------------------------------------------
# GLADO SESSION BANNER
# ------------------------------------------------------------

glado_banner() {
    local host_name="${HOSTNAME%%.*}"
    local shell_name="${SHELL##*/}"
    local user_name="${USER:-$(whoami)}"

    local mode="neural interface :: online"

    # SSH
    if [[ -n "$SSH_CONNECTION" ]]; then
        mode="remote shell :: established"
    fi

    # Root
    if [[ "$EUID" -eq 0 ]]; then
        mode="privileged shell :: active"
    fi

    # Terminal
    printf '\n'
    printf '╭─[ %-38s ]╮\n' "$host_name"
    printf '│  GLADO // neural shell                  │\n'
    printf '│  %-39s │\n' "$mode"
    printf '╰──────────────────────────────────────────╯\n'
    printf '\n'

    printf '  host       %s\n' "$host_name"
    printf '  shell      %s\n' "$shell_name"
    printf '  user       %s\n' "$user_name"
    printf '  directory  %s\n' "$PWD"

    if git rev-parse --is-inside-work-tree >/dev/null 2>&1; then
        printf '  git        %s\n' "$(git branch --show-current)"
    else
        printf '  git        —\n'
    fi

    printf '\n'
}

# Ejecutar solamente en una sesión interactiva
if [[ -o interactive ]]; then
    glado_banner
fi
```
