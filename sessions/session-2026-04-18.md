import React, { useState, useEffect } from 'react';
import {
  View, Text, StyleSheet, TouchableOpacity, Modal, ScrollView, ActivityIndicator,
} from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { COLORS, SPACING, RADIUS, SHADOWS } from '../utils/theme';
import { routeCalculator, CalculatedRoute } from '../utils/routeCalculator';
import { api } from '../utils/api';

const PDS_STATIONS = [
  'champery', 'les_crosets', 'champoussin', 'morgins', 'torgon',
  'avoriaz', 'morzine', 'les_gets', 'chatel', 'abondance', 'la_chapelle', 'st_jean_aulps',
];

interface RoutePlannerProps {
  visible: boolean;
  onClose: () => void;
  onRouteCalculated?: (route: CalculatedRoute) => void;
  stations: any[];
}

export default function RoutePlanner({ visible, onClose, onRouteCalculated, stations }: RoutePlannerProps) {
  const [originId, setOriginId] = useState('champery');
  const [destinationId, setDestinationId] = useState('chatel');
  const [calculating, setCalculating] = useState(false);
  const [route, setRoute] = useState<CalculatedRoute | null>(null);
  const [initialized, setInitialized] = useState(false);
  const [showOriginDropdown, setShowOriginDropdown] = useState(false);
  const [showDestDropdown, setShowDestDropdown] = useState(false);

  const pdsStations = stations.filter(s => PDS_STATIONS.includes(s.station_id));

  // Initialize route calculator on mount
  useEffect(() => {
    if (visible && !initialized) {
      initializeCalculator();
    }
  }, [visible]);

  const initializeCalculator = async () => {
    try {
      const allPistes: any[] = [];
      const allLifts: any[] = [];

      for (const stationId of PDS_STATIONS) {
        try {
          const p = await api.getPistes(stationId);
          const l = await api.getLifts(stationId);
          allPistes.push(...p);
          allLifts.push(...l);
        } catch {}
      }

      routeCalculator.initialize(allPistes, allLifts, pdsStations);
      setInitialized(true);
    } catch (e) {
      console.error('Error initializing route calculator:', e);
    }
  };

  const calculateRoute = () => {
    setCalculating(true);
    setTimeout(() => {
      const result = routeCalculator.calculate(originId, destinationId);
      setRoute(result);
      if (onRouteCalculated) {
        onRouteCalculated(result);
      }
      setCalculating(false);
    }, 300);
  };

  const getStationName = (id: string) => {
    return pdsStations.find(s => s.station_id === id)?.name || id;
  };

  return (
    <Modal
      visible={visible}
      transparent
      animationType="slide"
      onRequestClose={onClose}
    >
      {/* Overlay */}
      <View style={styles.overlay} />

      {/* Bottom Sheet */}
      <View style={styles.container}>
        {/* Handle Bar */}
        <View style={styles.handleBar} />

        {/* Header */}
        <View style={styles.header}>
          <Text style={styles.title}>Calculateur d'itinéraires</Text>
          <TouchableOpacity onPress={onClose} style={styles.closeBtn}>
            <Ionicons name="close" size={24} color={COLORS.text} />
          </TouchableOpacity>
        </View>

        <ScrollView style={styles.content} showsVerticalScrollIndicator={false}>
          {/* Origin Selector */}
          <Text style={styles.label}>Départ</Text>
          <TouchableOpacity
            style={styles.selector}
            onPress={() => setShowOriginDropdown(!showOriginDropdown)}
          >
            <Ionicons name="flag" size={18} color={COLORS.accent} />
            <Text style={styles.selectorText}>{getStationName(originId)}</Text>
            <Ionicons name={showOriginDropdown ? 'chevron-up' : 'chevron-down'} size={18} color={COLORS.textSecondary} />
          </TouchableOpacity>

          {showOriginDropdown && (
            <View style={styles.dropdown}>
              {pdsStations.map(s => (
                <TouchableOpacity
                  key={s.station_id}
                  style={[styles.dropdownItem, s.station_id === originId && styles.dropdownItemActive]}
                  onPress={() => {
                    setOriginId(s.station_id);
                    setShowOriginDropdown(false);
                  }}
                >
                  <Text style={[styles.dropdownItemText, s.station_id === originId && styles.dropdownItemTextActive]}>
                    {s.name}
                  </Text>
                </TouchableOpacity>
              ))}
            </View>
          )}

          {/* Destination Selector */}
          <Text style={[styles.label, { marginTop: SPACING.md }]}>Arrivée</Text>
          <TouchableOpacity
            style={styles.selector}
            onPress={() => setShowDestDropdown(!showDestDropdown)}
          >
            <Ionicons name="location" size={18} color={COLORS.accent} />
            <Text style={styles.selectorText}>{getStationName(destinationId)}</Text>
            <Ionicons name={showDestDropdown ? 'chevron-up' : 'chevron-down'} size={18} color={COLORS.textSecondary} />
          </TouchableOpacity>

          {showDestDropdown && (
            <View style={styles.dropdown}>
              {pdsStations.map(s => (
                <TouchableOpacity
                  key={s.station_id}
                  style={[styles.dropdownItem, s.station_id === destinationId && styles.dropdownItemActive]}
                  onPress={() => {
                    setDestinationId(s.station_id);
                    setShowDestDropdown(false);
                  }}
                >
                  <Text style={[styles.dropdownItemText, s.station_id === destinationId && styles.dropdownItemTextActive]}>
                    {s.name}
                  </Text>
                </TouchableOpacity>
              ))}
            </View>
          )}

          {/* Calculate Button */}
          <TouchableOpacity
            style={[styles.calculateBtn, calculating && { opacity: 0.5 }]}
            onPress={calculateRoute}
            disabled={calculating || originId === destinationId}
          >
            {calculating ? (
              <ActivityIndicator size="small" color="#FFF" />
            ) : (
              <>
                <Ionicons name="navigate" size={18} color="#FFF" />
                <Text style={styles.calculateBtnText}>Calculer</Text>
              </>
            )}
          </TouchableOpacity>

          {/* Route Result */}
          {route && (
            <View style={styles.resultCard}>
              {route.isValid ? (
                <>
                  <View style={styles.resultHeader}>
                    <Text style={styles.resultTitle}>✅ Route trouvée</Text>
                    <Text style={styles.resultTime}>⏱️ {route.totalTime} min</Text>
                  </View>

                  {/* Steps */}
                  <View style={styles.stepsContainer}>
                    {route.steps.map((step, idx) => (
                      <View key={idx} style={styles.stepRow}>
                        <View style={styles.stepNumber}>
                          <Text style={styles.stepNumberText}>{idx + 1}</Text>
                        </View>
                        <View style={styles.stepContent}>
                          <Text style={styles.stepType}>
                            {step.type === 'lift' ? '🚠 Remonte' : '⛷️ Piste'}
                          </Text>
                          <Text style={styles.stepName}>{step.name}</Text>
                          {step.difficulty && (
                            <Text style={styles.stepDifficulty}>
                              Difficulté: {step.difficulty.toUpperCase()}
                            </Text>
                          )}
                        </View>
                        <View style={styles.stepTime}>
                          <Text style={styles.stepTimeText}>{step.duration}min</Text>
                        </View>
                      </View>
                    ))}
                  </View>
                </>
              ) : (
                <View style={styles.errorContainer}>
                  <Ionicons name="alert-circle" size={32} color={COLORS.alert} />
                  <Text style={styles.errorText}>{route.message || 'Aucune route trouvée'}</Text>
                </View>
              )}
            </View>
          )}

          <View style={{ height: 30 }} />
        </ScrollView>
      </View>
    </Modal>
  );
}
