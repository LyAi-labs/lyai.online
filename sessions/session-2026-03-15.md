import sqlalchemy

def get_daily_context(engine):
    """
    Extrae lo que pasó hoy en Cervell para que Aurelius y Claude tengan de qué hablar.
    """
    with engine.connect() as conn:
        # Consultamos los últimos eventos del audit_log

            "SELECT endpoint, status_code, duration_ms FROM audit_log "
            "WHERE created_at > NOW() - INTERVAL '24 hours' LIMIT 50"
        )).fetchall()
        
        # Consultamos el crecimiento del Índice Universal
