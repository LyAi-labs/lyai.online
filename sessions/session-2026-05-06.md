     from clip_interrogator import Config, Interrogator
     from fastapi import FastAPI, File, HTTPException, UploadFile
     from PIL import Image
     
     _CI: Interrogator | None = None
     
     def _build_interrogator() -> Interrogator:
         device = "cuda" if torch.cuda.is_available() else "cpu"
         cfg = Config(
             clip_model_name="ViT-L-14/openai",
             caption_model_name="blip-base",  # was blip_model_name (deprecated)
             device=device,
             cache_path="/opt/lyai/services/clip-interrogator/cache",
             clip_offload=(device == "cpu"),
             caption_offload=(device == "cpu"),
             quiet=True,
         )
         return Interrogator(cfg)
     
     @asynccontextmanager
     async def lifespan(app: FastAPI):
         global _CI
         _CI = _build_interrogator()
         yield
         _CI = None
     
     app = FastAPI(lifespan=lifespan)
     
     @app.get("/health")
     async def health():
         return {"ok": _CI is not None, "device": "cuda" if torch.cuda.is_available() else "cpu"}
     
     @app.post("/interrogate")
     async def interrogate(file: UploadFile = File(...), mode: str = "both"):
         if _CI is None: raise HTTPException(503, "interrogator not ready")
         raw = await file.read()
         img = Image.open(io.BytesIO(raw)).convert("RGB")
         out = {}
         t0 = time.time()
         if mode in ("fast", "both"):
             out["prompt_fast"] = _CI.interrogate_fast(img)
         if mode in ("best", "both"):
             if torch.cuda.is_available():
                 out["prompt_best"] = _CI.interrogate(img)
             else:
                 out["prompt_best"] = _CI.interrogate_classic(img)
         out["elapsed_s"] = round(time.time() - t0, 2)
         out["device"] = "cuda" if torch.cuda.is_available() else "cpu"
         out["model"] = "ViT-L/14 + BLIP-base"
         return out
